## 1. 拓扑排序 `n(e,t)` —— 逐节点还原

```js
function n(e,t){
  let a=new Set(e.filter(e=>"group"!==e.type).map(e=>e.id)),   // 节点集：★ 排掉 type==="group"
      r={},n={};
  for(let e of a) r[e]=0, n[e]=[];                              // r=入度表  n=出边邻接表
  for(let e of t) if(a.has(e.source)&&a.has(e.target))          // 只认两端都在节点集里的边
      r[e.target]=(r[e.target]||0)+1, n[e.source].push(e.target);
  let s=[], o=Object.keys(r).filter(e=>0===r[e]);               // 入度 0 的一层
  for(;o.length>0;){
    s.push([...o]);                                             // ★ 一层一层出，s 是「层数组」
    let e=[];
    for(let t of o) for(let a of n[t]||[]) { r[a]--; if(0===r[a]) e.push(a); }
    o=e;
  }
  if(s.reduce((e,t)=>e+t.length,0)<a.size) throw Error("工作流中存在循环依赖，无法执行");
  return {layers:s, totalSteps:a.size};
}
```

要点（照抄）：
- 返回 **分层**（`layers`），不是一条扁平的顺序 —— 同层可以并发。
- `type === "group"` 的节点**根本不进图**（组只是视觉容器）。
- 孤立节点（没有任何边）入度 0 → 自成第一层，**不会被丢掉**。
- 环检测 = 「出过的节点总数 < 节点总数」→ 抛 `工作流中存在循环依赖，无法执行`。
- 有向边方向：`source` → `target`，入度加在 `target` 上。

## 2. 端口与数据键

```js
let s=/^out-n\d+-/;                       // 「多镜头」动态出端口前缀
function o(e){return (e.data.outputs||[]).some(e=>s.test(e.id))}   // 这个节点是不是动态口
function i(e){return !!e && s.test(e)}                             // 这个端口 id 是不是动态口

// 给一条边解析出「下游用哪个输入口 / 写进 upstreamData 的哪个键」
function l(e,t,a){
  let r = "string"==typeof e.data?.targetPort && e.data.targetPort ? e.data.targetPort : e.targetHandle,
      n = t.find(e=>e.id===r) ?? (a ? s(a,t) : void 0);
  return {port:n, dataKey:n?.id || r || "input"};
}

// 一个节点的「输出值」（按优先级抠）
function i(e){
  return e.result?.outputUrl || e.params?.file || e.params?.content
      || ("builtin:text-input"===e.moduleId ? e.params?.prompt : "") || "";
}

// 把上游边的输出收集成 upstreamData
function d(e,t,a){                       // e=入边们 t=id→节点 a=本节点 inputs
  let n={};
  for(let s of e){
    let e=t.get(s.source); if(!e) continue;
    let o=i(e.data);       if(!o) continue;                        // 上游没值 → 跳过
    let d=e.data.outputs[0]?.dataType||"any",
        c=l(s,a,d),
        u=c.port ?? r(d,a);                                        // r() 是按类型找兼容口
    n[c.port ? c.dataKey : (u?.id||c.dataKey)] = o;
  }
  return n;
}
```

- **边的字段名（实测）**：`{id, source, target, sourceHandle, targetHandle, data:{sourcePort?, targetPort?}}`。
  `sourceHandle`/`targetHandle` 是 React Flow 标准字段；`data.sourcePort`/`data.targetPort` 是飞彩的覆盖字段，**有就优先**。
- 下游写进 `data.upstreamData[dataKey]`；`dataKey` 默认 = 目标口 id，没有口就退 `"input"`。
- 写回下游的差异比较用 `cT`（浅比较两个对象键与值），一致就不写 —— 少一次重渲染。

## 3. 执行主循环 `m(...)` 的硬约束（实测字符串）

| 约束 | 实测字符串 |
|---|---|
| 并发上限 | `u=30`；`x=i.maxConcurrency??u`；`if(!Number.isSafeInteger(x)||x<1||x>u) throw Error(\`画布执行并发必须是 1-${u} 的整数\`)` |
| 截止时间 | `if(void 0!==i.deadlineAt&&(!Number.isSafeInteger(i.deadlineAt)||i.deadlineAt<=0)) throw Error("画布执行截止时间无效")` |
| 时钟 | `if(!Number.isFinite(e)) throw Error("画布执行时钟无效")` |
| 启动门禁 | `beforeNodeStart` 返回 false → `"节点启动门禁已拒绝"` |
| 上游失败传染 | `` `上游 ${u.source.slice(-6)} 失败,已跳过${a?`：${a}`:""}` ``（`a` 是上游错误前 80 字，超长加 `…`） |
| 迟到结果 | `"节点执行结果已与当前队列或画布失配，迟到结果未写回"`（`shouldCommitNodeResult` 返回 false） |
| 未启动 | `class p extends Error { this.name="CanvasNodeNotStartedError" }`，消息 `"节点未启动"` |

- `b = o ? 1 : x` —— 一旦进入「收尾/串行」模式（`o`）就压成 **1**。
- 执行前先**快照每个节点的 `upstreamData`** 到 `k`，失败重跑时用它回滚输入。
- 失败节点单独重跑：`f.push(e); g.add(e)`（记失败集 + 跳过集），下游节点在 `a.filter(t=>t.target===e).find(e=>j.has(e.source))` 命中时直接判「上游失败，已跳过」。
- 动态端口续接：`n.upstreamInjections` 允许把上游结果**额外注入**到别的节点（多镜头的每镜图）。

## 4. 连线合法性 `r5(e,t,r)`（e=节点表 t=源 r=目标）

```js
let a=e.find(e=>e.nodeId===t.nodeId), n=e.find(e=>e.nodeId===r.nodeId);
if(!a||!n||a.nodeId===n.nodeId
   ||!Object.hasOwn(r2,a.moduleId)||!Object.hasOwn(r2,n.moduleId)) return !1;

    o=i[t.portId], l=s[r.portId];
```

四条：① 两端节点必须在；② **不许自连**（`a.nodeId===n.nodeId`）；③ 两端模块都必须在端口契约表 `r2` 里，**不在就一律拒绝**；④ 判据就是 **出端口 dataType === 入端口 dataType**（严格相等，没有 `any` 兜底）。

## 6. 画布改动提案契约（approval contract）

来自 `7528...js @214034-218958`（`existingCanvasMutation` 校验器）。每条提案的公共骨架：

```js
{schemaVersion:1, costCents, approvalRequired, runNodes, evidence, mutation}
```

- `approvalRequired` 必须 `!0`
- `costCents` 必须是 safe integer，`0 <= costCents <= 800`
- `evidence` 必须有 `canvasRevision`（`/^[0-9a-f]{64}$/`）、`evidenceDigest`、`selectionInstanceId`、`selectionVersion`、`node(s)`
- `mutation.kind` 枚举：**`set_params` / `connect` / `auto_layout` / `create_reference_extractors`**
- **这四种 mutation 全部要求 `costCents === 0` 且 `runNodes === false`** —— 改画布本身永远不花钱
- `connect` 的 mutation 还必须**当场过 `r5`**（见 §4），且 source/target 的 nodeId 排序后必须等于 evidence 里的节点顺序
- `auto_layout` 要求 `nodeIds` 与 `positions` 一一对应（`{nodeId,x,y}` 有限数），至少 2 个节点，最多 64
- `create_reference_extractors` 要求 `sourceNodeIds` 的模块全是 `builtin:text-input`，`runAfterCreate` 是 bool，
  `estimatedCents === (runAfterCreate ? a : 0)`，且 `runNodes === mutation.runAfterCreate`、`costCents === mutation.estimatedCents`
- `canvasPromptEdit`（改一个节点提示词）文案逐字：**「修改当前选中的一个画布节点提示词，零费用，等待用户批准」**，
  要求 `costCents===0`、`runNodes===false`、`prompt` 非空且 ≤20000 字节
